const SHEET_ID = '1uVfo5LtFLbiYoIXz4B70ZkPOHEBIPek9e0y2XKv4HZk'; function getSheetData() { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName("Sheet1"); if (!sheet) { sheet = ss.getSheetByName("Main CIG LIST"); } const data = sheet.getDataRange().getValues(); const items = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[0] || row[0].toString().trim() === "") continue; items.push({ item_name: row[0].toString().trim(), item_number: row[1] ? row[1].toString().trim() : "", upc: row[2] ? row[2].toString().trim() : "", must_have: parseFloat(row[3]) || 0 }); } return items; } catch(e) { return [{error: e.toString()}]; } } function doGet(e) { try { // If no parameters, return the items list if (!e.parameter.date) { const items_list = getSheetData(); return ContentService.createTextOutput(JSON.stringify(items_list)) .setMimeType(ContentService.MimeType.JSON); } // Otherwise, log the order const date = e.parameter.date || ''; const week = e.parameter.week || ''; const items = e.parameter.items || ''; if (date && week && items) { logOrderToSheet(date, week, items); return ContentService.createTextOutput(JSON.stringify({success: true, message: 'Order logged successfully'})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({success: false, message: 'Missing parameters'})) .setMimeType(ContentService.MimeType.JSON); } catch(e) { return ContentService.createTextOutput(JSON.stringify({error: e.toString(), success: false})) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const params = e.parameter; const date = params.date || ''; const week = params.week || ''; const items = params.items || ''; if (date && week && items) { logOrderToSheet(date, week, items); return ContentService.createTextOutput(JSON.stringify({success: true, message: 'Order logged successfully'})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({success: false, message: 'Missing parameters'})) .setMimeType(ContentService.MimeType.JSON); } catch(e) { return ContentService.createTextOutput(JSON.stringify({error: e.toString(), success: false})) .setMimeType(ContentService.MimeType.JSON); } } function logOrderToSheet(date, week, itemsString) { const ss = SpreadsheetApp.openById(SHEET_ID); // Create Order Summary sheet if it doesn't exist let summarySheet = ss.getSheetByName("Order Summary"); if (!summarySheet) { summarySheet = ss.insertSheet("Order Summary", 0); summarySheet.appendRow(['Date', 'Week', 'Item Count', 'Total Quantity', 'Timestamp']); } // Create Order Details sheet if it doesn't exist let detailsSheet = ss.getSheetByName("Order Details"); if (!detailsSheet) { detailsSheet = ss.insertSheet("Order Details", 1); detailsSheet.appendRow(['Date', 'Week', 'Item Name', 'Item #', 'Qty', 'Timestamp']); } // Parse items string: "Marlboro Red Short (#4336145): 5 | Newport Box (#4340626): 6" const itemsArray = itemsString.split(' | ').filter(item => item.trim()); let totalQty = 0; const timestamp = new Date(); // Add summary row FIRST summarySheet.appendRow([date, week, itemsArray.length, '', timestamp]); const summaryRow = summarySheet.getLastRow(); // Add detail rows for each item itemsArray.forEach(itemStr => { // Parse format: "Marlboro Red Short (#4336145): 5" const match = itemStr.match(/(.+?)\s*\(#(\d+)\):\s*([\d.]+)/); if (match) { const itemName = match[1].trim(); const itemNum = match[2]; const qty = parseFloat(match[3]); totalQty += qty; detailsSheet.appendRow([date, week, itemName, itemNum, qty, timestamp]); } }); // Update total quantity in summary summarySheet.getRange(summaryRow, 4).setValue(totalQty); // Force sheet recalculation ss.toast('✅ Order logged: ' + itemsArray.length + ' items, Total: ' + totalQty); }
