async function logToSheets() { const button = document.getElementById('logButton'); const ordered = getOrderedItems(); if (ordered.length === 0) { alert('❌ No items with positive orders'); return; } console.log('💾 Logging order to sheets via Apps Script...'); button.disabled = true; button.innerHTML = '
Saving...'; try { const today = new Date(); const weekNum = getWeekNumber(today); const dateStr = today.toLocaleDateString('en-US', { year: 'numeric', month: '2-digit', day: '2-digit' }); // Format: "Marlboro Red Short (#4336145): 5 | Newport Box (#4340626): 6" const ordersText = ordered.map(item => `${item.name} (#${item.itemNumber}): ${item.qty}` ).join(' | '); // Call Apps Script doPost endpoint const appsScriptURL = 'https://script.google.com/macros/s/AKfycbxP5W9BDfiXLvNAq5Uk99G4a0qe7WQ9UD2-Iz-ir-3sgOVVG_B155vCxwr-DpqGWhMzUA/exec'; const formData = new FormData(); formData.append('date', dateStr); formData.append('week', weekNum); formData.append('items', ordersText); const response = await fetch(appsScriptURL, { method: 'POST', body: formData }); const result = await response.text(); console.log('✅ Apps Script response:', result); if (result.includes('success')) { showSuccessPopup(); setTimeout(() => { resetForm(); button.disabled = false; button.innerHTML = '💾 Log to Sheet'; }, 2000); } else { throw new Error('Apps Script returned error'); } } catch (error) { console.error('❌ Error logging order:', error); alert('❌ Error logging order: ' + error.message); button.disabled = false; button.innerHTML = '💾 Log to Sheet'; } }
